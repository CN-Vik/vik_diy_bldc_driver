```c
#include <math.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846f
#endif

typedef struct {
    // 硬件与控制参数
    float Rs;          // 定子电阻 (Ω)
    float Ls;          // 定子电感 (H)
    float Ts;          // 采样周期 (s)
    float k_smo;       // 滑模控制增益
    float lpf_alpha;   // 低通滤波系数 (Ts * 2 * pi * f_cutoff)

    // 内部状态变量
    float i_alpha_hat; // alpha 轴估计电流 (A)
    float i_beta_hat;  // beta 轴估计电流 (A)
    float e_alpha;     // 滤波后的 alpha 轴反电势 (V)
    float e_beta;      // 滤波后的 beta 轴反电势 (V)

    // 输出结果
    float theta_e;     // 输出电角度 (rad, [0, 2*PI])
} SMO_Controller;

/**
 * @brief 初始化滑模观测器
 */
void SMO_Init(SMO_Controller *smo, float Rs, float Ls, float Ts, float k_smo, float lpf_cutoff_hz) {
    smo->Rs = Rs;
    smo->Ls = Ls;
    smo->Ts = Ts;
    smo->k_smo = k_smo;
    smo->lpf_alpha = Ts * 2.0f * M_PI * lpf_cutoff_hz;
    smo->i_alpha_hat = 0.0f;
    smo->i_beta_hat = 0.0f;
    smo->e_alpha = 0.0f;
    smo->e_beta = 0.0f;
    smo->theta_e = 0.0f;
}

/**
 * @brief 饱和函数 (替代 Sign 极性函数，降低高频抖振)
 */
static inline float SMO_Sat(float err, float delta) {
    if (err > delta) return 1.0f;
    if (err < -delta) return -1.0f;
    return err / delta;
}

/**
 * @brief 滑模观测器更新 API
 * @param smo 控制器句柄
 * @param u_alpha alpha 轴电压 (V)
 * @param u_beta  beta 轴电压 (V)
 * @param i_alpha alpha 轴实际采样电流 (A)
 * @param i_beta  beta 轴实际采样电流 (A)
 * @return float  返回当前计算得到的电角度 (rad)
 */
float SMO_Update(SMO_Controller *smo, float u_alpha, float u_beta, float i_alpha, float i_beta) {
    // 1. 计算电流估计误差
    float i_alpha_err = smo->i_alpha_hat - i_alpha;
    float i_beta_err  = smo->i_beta_hat - i_beta;

    // 2. 切换控制律信号 z (使用饱和函数平滑控制)
    float z_alpha = smo->k_smo * SMO_Sat(i_alpha_err, 0.5f);
    float z_beta  = smo->k_smo * SMO_Sat(i_beta_err, 0.5f);

    // 3. 一阶离散化更新电流估计值: di/dt = (-Rs/Ls)*i + (u - z)/Ls
    smo->i_alpha_hat += smo->Ts * ((-smo->Rs / smo->Ls) * smo->i_alpha_hat + (u_alpha - z_alpha) / smo->Ls);
    smo->i_beta_hat  += smo->Ts * ((-smo->Rs / smo->Ls) * smo->i_beta_hat + (u_beta - z_beta) / smo->Ls);

    // 4. 一阶低通滤波器提取反电动势 e = LPF(z)
    smo->e_alpha += smo->lpf_alpha * (z_alpha - smo->e_alpha);
    smo->e_beta  += smo->lpf_alpha * (z_beta - smo->e_beta);

    // 5. 反正切求电角度: e_alpha = -E*sin(theta), e_beta = E*cos(theta)
    // 故 tan(theta) = -e_alpha / e_beta
    smo->theta_e = atan2f(-smo->e_alpha, smo->e_beta);

    // 6. 将弧度约束在 [0, 2*PI] 范围
    if (smo->theta_e < 0.0f) {
        smo->theta_e += 2.0f * M_PI;
    }

    return smo->theta_e;
}

```

**关键推导与设计说明**

* **反电势建模逻辑**：永磁同步电机（PMSM）在静止坐标系下的反电势公式为 $e_\alpha = -E\sin\theta_e$，$e_\beta = E\cos\theta_e$。利用反正切函数估算电角度时，关系式为 $\theta_e = \text{atan2}(-e_\alpha, e_\beta)$。
* **抖振抑制**：采用 `SMO_Sat`（饱和函数）替代传统的 `sign` 符号函数，在误差极小区域实现线性连续切换，显著降低开关引起的抖振。
* **相位延迟补偿（工程使用注意）**：低通滤波器 `LPF` 在提取 $e_\alpha, e_\beta$ 时会引入相位滞后 $\Delta \theta = \arctan(\omega_e / \omega_{cutoff})$。实际在高速运行时，需在返回 `theta_e` 后根据当前转速补偿该延迟角度。

/**
 * @brief 饱和函数 (Saturation Function)
 * @details 替代传统滑模控制中的符号函数 (Sign)，在边界层 [-delta, delta] 内进行线性过渡，
 *          超出边界层输出 ±1.0。可消除高频开关动作带来的高频抖振 (Chattering)。
 * 
 * @param err   控制误差输入 (如电流估计误差 i_hat - i)
 * @param delta 边界层厚度 (Boundary Layer)，决定线性区的半宽 (要求 > 0)
 * @return float 归一化的控制输出信号，取值范围 [-1.0f, 1.0f]
 */
static inline float SMO_Sat(float err, float delta) 
{
    if (err > delta) return 1.0f;
    if (err < -delta) return -1.0f;
    return err / delta;
}

问到点子上了！$k_{smo}$ 的确定是滑模观测器（SMO）调参中最核心的一环。

确定 $k_{smo}$ 的核心原则：**$k_{smo}$ 的取值必须严格大于电机运行过程中的最大反电动势 (Back-EMF) 幅值**，通常设定为 **$1.2 \sim 2.0$ 倍的安全裕量**。

---

### 1. 数学理论依据（为什么必须大于反电动势？）

滑模控制需要满足李雅普诺夫可达性条件（即误差必须收敛 $S \cdot \dot{S} < 0$）。

在电机模型中，电流估计误差的微分方程展开后，滑模信号 $z$ 必须能够“覆盖”住反电动势 $e$。推导出的充要条件为：


$$k_{smo} > \max(\vert{}e_\alpha\vert{}, \vert{}e_\beta\vert{}) = E$$

其中，反电动势幅值 $E$ 随转速成正比：


$$E = \omega_e \cdot \psi_f$$

* $\omega_e$：电角速度 $(\text{rad/s})$
* $\psi_f$：永磁体磁链 $(\text{Wb})$

---

### 2. 工程确定 $k_{smo}$ 的三种实用方法

#### 方法一：理论估算法（固定增益，最常用）

1. **计算电机最高转速下的反电动势 $E_{max}$**：

$$E_{max} = \left( \frac{n_{max} \cdot 2\pi}{60} \cdot P_p \right) \cdot \psi_f$$


* $n_{max}$：设计最高转速 $(\text{RPM})$
* $P_p$：电机极对数
* $\psi_f$：磁链（可以通过电机反电动势系数 $K_e$ 推算，或查阅电机参数表）


2. **设定 $k_{smo}$**：

$$k_{smo} = (1.2 \sim 1.5) \cdot E_{max}$$



* **缺点**：低速时 $E$ 很小，但 $k_{smo}$ 保持极大，会导致低速时高频噪声和抖振非常严重。

---

#### 方法二：自适应增益法（性能最好，工业级推荐）

为了解决“低速抖振大、高速不够用”的矛盾，可以将 $k_{smo}$ 写成**转速的函数**，让增益随转速动态变化：

$$k_{smo} = k_{base} + k_{gain} \cdot \vert{}\omega_e\vert{}$$

* $k_{base}$：基础增益，保证极低速时的基本响应（通常取额定电压的 5%~10%）
* $k_{gain}$：斜率系数，满足 $k_{gain} > \psi_f$ 即可（通常取 $1.2 \times \psi_f \sim 1.5 \times \psi_f$）

---

#### 方法三：实机台架调试法（没有磁链参数时）

如果你不知道电机的磁链 $\psi_f$ 参数，可以在台架上按以下步骤试凑：

1. **设定初始值**：先设为母线电压 $V_{dc}$ 的 **20%~30%**（例如 24V 母线先填 5.0）。
2. **带载拉速测试**：
* 将电机逐步加速到目标最高转速，同时突加负载。
* **现象判断**：
* **$k_{smo}$ 太小**：在高速或突然加载时，估计电流 $\hat{i}$ 无法跟踪实际电流 $i$，观测器失去响应，算出来的角度直接紊乱、电机失步发散。
* **$k_{smo}$ 太大**：电流跟踪没问题，但输出的反电动势噪声/高频毛刺极多，即使经过低通滤波（LPF）角度依然剧烈抖动，甚至引起电机噪音。




3. **锁定最佳值**：找到能够保证**最高转速 + 满载不发散**的最小 $k_{smo}$ 值，然后将其乘以 **1.2** 作为最终稳健系数。

---

### 调试经验总结

| 参数表现 | 现象/影响 | 解决办法 |
| --- | --- | --- |
| **$k_{smo}$ 过小** | 高速/重载下角度观测发散、失步、过流 | 增大 $k_{smo}$ |
| **$k_{smo}$ 过大** | 观测器噪声大、高频啸叫、角度抖动 | 减小 $k_{smo}$，或改用**自适应增益** |

这里关键要区分 C 标准库中的 **`atan()`** 和 **`atan2f()`** 两个函数。你的担心对 `atan()` 成立，但 **`atan2f()` 恰恰就是为了解决这个 0 ~ 2π 全象限问题而设计的！**

---

### 1. `atan()` 与 `atan2f()` 的本质区别

* **`atan(y/x)`（单参数普通反正切）**：
* 输入是一个除法比值。
* **问题**：比值会丢失符号信息。例如 $y=1, x=1$（第一象限）和 $y=-1, x=-1$（第三象限），计算出来的 $y/x$ 都是 $1.0$，`atan(1.0)` 只能返回 $\frac{\pi}{4}$ ($45^\circ$)。
* **值域**：仅为 **$(-\frac{\pi}{2}, \frac{\pi}{2})$**（即 $-90^\circ \sim +90^\circ$）。


* **`atan2f(y, x)`（双参数四象限反正切）**：
* 输入是**独立的分子 $y$ 和分母 $x$**，不提前做除法。
* **原理**：内部会同时判断 $y$ 和 $x$ 各自的正负号，从而精准确定点在平面笛卡尔坐标系的**哪一个象限**（第一到第四象限）。
* **值域**：包含了完整圆周的 **$(-\pi, \pi]$**（即 $-180^\circ \sim +180^\circ$）。



---

### 2. 角度是如何映射到 $[0, 2\pi]$ 的？

`atan2f` 计算出的初始结果范围是 $(-\pi, \pi]$，配合后文的 `+ 2*FOC_PI` 逻辑，就能完美平滑过渡到 $[0, 2\pi]$ ($0 \sim 6.28$)：

| 分子 $y$ (Alpha) | 分母 $x$ (Beta) | 所在象限 | `atan2f` 原始返回值 | 执行 `if (<0) += 2π` 后 | 对应角度 (度) |
| --- | --- | --- | --- | --- | --- |
| $+1$ | $+1$ | **第一象限** | $+0.25\pi$ | $+0.25\pi$ | $45^\circ$ |
| $+1$ | $-1$ | **第二象限** | $+0.75\pi$ | $+0.75\pi$ | $135^\circ$ |
| $-1$ | $-1$ | **第三象限** | $-0.75\pi$ | **$-0.75\pi + 2\pi = 1.25\pi$** | $225^\circ$ |
| $-1$ | $+1$ | **第四象限** | $-0.25\pi$ | **$-0.25\pi + 2\pi = 1.75\pi$** | $315^\circ$ |

---

### 3. 代码写法注意事项

对于反电动势模型：


$$e_\alpha = -E \sin\theta_e$$

$$e_\beta = E \cos\theta_e$$

计算 $\tan\theta_e = \frac{\sin\theta_e}{\cos\theta_e} = \frac{-e_\alpha}{e_\beta}$ 时，以下两种写法在数学和工程上都是完全等价且正确的：

```c
// 写法 A：将负号放入 y 参数（推荐，直观表示 sin/cos 的比值）
smo->theta_e = atan2f(-smo->ebmf_alpha, smo->ebmf_beta);

// 写法 B：利用奇函数性质将负号提在外面（即你贴出的代码）
smo->theta_e = -atan2f(smo->ebmf_alpha, smo->ebmf_beta);

```

只需要接上你代码里的两行：

```c
if (smo->theta_e < 0.0f) {
    smo->theta_e += 2.0f * FOC_PI; // 将 (-PI, 0) 映射到 (PI, 2PI)
}

```

输出的 `theta_e` 就会是标准的 $0 \sim 2\pi$ ($0 \sim 6.28\text{ rad}$) 连续电角度。
